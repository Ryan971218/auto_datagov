  f.gmv - w.gmv AS diff,

FROM finance_gmv f
JOIN warehouse_gmv w USING (month)
WHERE ABS(f.gmv - w.gmv) > 100  -- 阈值 100 元
ORDER BY 1 DESC
LIMIT 10;

-- 财务 GMV vs 数仓 GMV 月度对账
-- 来源: concepts/gmv.md, sops/sop-gmv-001
WITH finance_gmv AS (
  SELECT
    DATE_TRUNC('month', order_date) AS month,
    SUM(paid_amount_cny) - SUM(refund_amount_cny) AS gmv
  FROM finance.dm_finance_gmv
  WHERE order_date >= DATE_TRUNC('month', CURRENT_DATE - INTERVAL '1 month')
  GROUP BY 1
),
warehouse_gmv AS (
  SELECT
    DATE_TRUNC('month', paid_at) AS month,
    SUM(paid_order_amount) - SUM(refund_amount) AS gmv
  FROM {{ ref('dm_finance_gmv') }}
  WHERE paid_at >= DATE_TRUNC('month', CURRENT_DATE - INTERVAL '1 month')
  GROUP BY 1
)
SELECT
  f.month,
  f.gmv AS finance_gmv,
  w.gmv AS warehouse_gmv,
  f.gmv - w.gmv AS diff,
  ROUND((f.gmv - w.gmv) * 100.0 / f.gmv, 2) AS diff_pct
FROM finance_gmv f
JOIN warehouse_gmv w USING (month)
WHERE ABS(f.gmv - w.gmv) > 100  -- 阈值 100 元
ORDER BY 1 DESC;
⚠ 需要你确认：表名 finance.dm_finance_gmv 和字段 paid_amount_cny 是我推断的，不一定是你项目的真实表。请检查后使用。
